# Runs of the Diffusion case, by scripts/test_cases.sh and test_cases.py. A line "[LABEL] ARGS"
# starts the base run LABEL, built with "make ARGS" and run with "<exe> FILE" for inputs=FILE in
# ARGS; the variant lines below it (a name and ParmParse overrides appended to that command
# line) use the same executable. The labels are the names the runs had as separate cases.
# Sections with the same BUILD share one build; prob.problem in each inputs file picks the
# problem.

# ===== Two-layer Couette flow (prob.problem = Couette2Layer) =====

[Couette2Layer-5Eq]          BUILD=5Eq     inputs=prob/inputs
# The base inputs use the five-equation model with DIFFUSION = true (the reference of the
# six-equation runs), a sharp interface, the viscosity ratio mu_2/mu_1 = 10 and MUSCL-MC, to the
# steady state.
# --- Viscosity ratio 0.1 (phase 2 the less viscous), and a diffuse interface ---
ratio-01                max_step=20 prob.EOS.mu_1=0.1 prob.EOS.mu_2=0.01 prob.rho_1=10.0 prob.rho_2=1.0
diffuse                 max_step=20 PhaseField.Eta_Multiplier=1.0
# --- Scheme (three ghost cells through the user wall condition) ---
scheme-weno5            max_step=20 FiniteVolume.Scheme=WENO5

[Couette2Layer-6Eq]          BUILD=6Eq     inputs=prob/inputs.Couette2Layer-6Eq
# The base inputs use the two-phase six-equation model with DIFFUSION = true, a sharp interface,
# the viscosity ratio mu_2/mu_1 = 10, MUSCL-MC and pressure relaxation, to the steady state.
# --- Viscosity ratio 0.1 (phase 2 the less viscous), a diffuse interface, and a sharp interface
# in the middle of a cell (one cell with alpha = 1/2, which keeps the first-order error of the
# mixture viscosity: the harmonic face mean is exact only for an interface on a cell face) ---
ratio-01                max_step=20 prob.EOS.mu_1=0.1 prob.EOS.mu_2=0.01 prob.rho_1=10.0 prob.rho_2=1.0
diffuse                 max_step=20 PhaseField.Eta_Multiplier=1.0
mid-cell                max_step=20 prob.h=0.515625
# --- Scheme (three ghost cells through the user wall condition) and relaxation ---
scheme-weno5            max_step=20 FiniteVolume.Scheme=WENO5
no-relaxation           max_step=20 Physics.pressure_relaxation=0

[Couette2Layer-6Eq-N]        BUILD=6Eq-N   inputs=prob/inputs.Couette2Layer-6Eq-N
# Joint equilibrium can be selected independently of the flux formulation.
relax-pt                max_step=20 Physics.Relaxation=PressureTemperature
# Identical caloric gases: dispatch/parser regression; nontrivial mass transfer is checked natively.
chemical-identical      max_step=5 Physics.Relaxation=PressureTemperature Physics.ChemicalRelaxation=1 Physics.ChemicalRelaxation.Pairs=0 1 Physics.ChemicalRelaxation.Tolerance=1e-9 Physics.ChemicalRelaxation.MaxSweeps=128 prob.EOS.Substance=gas gas prob.EOS.sref=0 0 prob.EOS.T_ref=1 prob.EOS.p_ref=1
compatible              max_step=1000000 Physics.CompatibleDiffusion=1 Physics.Relaxation=PressureTemperature
# Current IE solver options; the final pressure/energy reset stays on in both.
hllc-consistent        max_step=10 Physics.HLLCImplementation=Consistent
# The base inputs use the N-phase six-equation model (NPHASE = 2) with DIFFUSION = true, a sharp
# interface, the viscosity ratio mu_2/mu_1 = 10, MUSCL-MC and pressure relaxation, to the steady
# state.
# --- Viscosity ratio 0.1 (phase 2 the less viscous), and a diffuse interface ---
ratio-01                max_step=20 prob.EOS.mu=0.1 0.01 prob.rho=10.0 1.0
diffuse                 max_step=20 PhaseField.Eta_Multiplier=1.0
# --- Scheme (three ghost cells through the user wall condition) and relaxation ---
scheme-weno5            max_step=20 FiniteVolume.Scheme=WENO5

# Interface inside a cell: distinguish interface error from face-aligned traction.
mid-cell                max_step=200 prob.h=0.515625

sharp                   max_step=20 PhaseField.Eta_Multiplier=0.0

[HeatConduction2Mat-6Eq]     BUILD=6Eq     inputs=prob/inputs.HeatConduction2Mat-6Eq

# Current IE solver options; the final pressure/energy reset stays on in both.
hllc-consistent        max_step=10 Physics.HLLCImplementation=Consistent

# --- A sharp interface (a step in alpha), and no relaxation ---
